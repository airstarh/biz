public function test_hold_creation_returns_409_when_slot_full()
{
    $slot = Slot::factory()->create(['remaining' => 0]);

    $response = $this->withHeaders(['Idempotency-Key' => 'uuid-123'])
                     ->postJson("/api/v1/slots/{$slot->id}/hold");

    $response->assertStatus(409)
             ->assertJson(['error' => 'No available slots']);
}
